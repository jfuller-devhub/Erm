-- ── ERM_RISK_ASSESSMENT ───────────────────────────────────────
-- Scored evaluation of a risk at a point in time.
-- Scores: LIKELIHOOD_SCORE (1-5) × IMPACT_SCORE (1-5) → INHERENT_SCORE (1-25).
-- RESIDUAL_SCORE = after existing controls.
-- TARGET_SCORE   = desired end-state.
-- IS_CURRENT     = TRUE for the active assessment; application
--                  layer enforces at most one current per risk.
-- localStorage key: 'erm_risk_assessments_v1'

CREATE TABLE ERM_RISK_ASSESSMENT (
  RISK_ASSESSMENT_ID   INT          NOT NULL AUTO_INCREMENT PRIMARY KEY COMMENT 'Surrogate primary key for the assessment record',
  CODE                 VARCHAR(20)  NOT NULL                            COMMENT 'Human-readable identifier, e.g. ASMT-001; unique across all risk assessments',
  RISK_ID              INT          NOT NULL                            COMMENT 'Reference to the risk being assessed; FK to ERM_RISK',
  REVIEWER_APP_USER_ID INT          NULL                                COMMENT 'Internal user who reviewed and approved this assessment; FK to ERM_APP_USER',
  ASSESSMENT_DATE      DATE         NOT NULL                            COMMENT 'Calendar date on which this risk assessment was formally conducted',
  ASSESSMENT_TYPE      VARCHAR(15)  NOT NULL DEFAULT 'periodic'         COMMENT 'Reason this assessment was performed (periodic, triggered, ad_hoc); see enum: assessment_type',  -- enum: assessment_type
  LIKELIHOOD_SCORE     TINYINT      NOT NULL                            COMMENT 'Probability of the risk occurring on a 1 (very unlikely) to 5 (almost certain) scale',
  IMPACT_SCORE         TINYINT      NOT NULL                            COMMENT 'Severity of consequences if the risk occurs on a 1 (negligible) to 5 (catastrophic) scale',
  VELOCITY_SCORE       TINYINT      NULL                                COMMENT 'Speed at which the risk could materialise on a 1 (slow) to 5 (immediate) scale; optional',
  INHERENT_SCORE       DECIMAL(6,2) NOT NULL                            COMMENT 'Raw risk exposure before controls, calculated as LIKELIHOOD_SCORE multiplied by IMPACT_SCORE',
  RESIDUAL_SCORE       DECIMAL(6,2) NOT NULL                            COMMENT 'Remaining risk exposure after applying existing controls; must be less than or equal to INHERENT_SCORE',
  TARGET_SCORE         DECIMAL(6,2) NULL                                COMMENT 'Desired residual score after all planned mitigations are fully implemented',
  RISK_RATING          VARCHAR(15)  NOT NULL                            COMMENT 'Descriptive band derived from RESIDUAL_SCORE (critical, high, medium, low, negligible); see enum: risk_rating', -- enum: risk_rating
  NOTES                TEXT         NULL                                COMMENT 'Assessor commentary, rationale for scores, or any caveats relevant to this assessment',
  IS_CURRENT           BOOLEAN      NOT NULL DEFAULT FALSE              COMMENT 'Flag indicating this is the currently active assessment for the risk; application enforces at most one per risk',
  CREATED_AT           DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP  COMMENT 'Timestamp at which this assessment record was created; set automatically on insert',
  UPDATED_AT           DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP
                                             ON UPDATE CURRENT_TIMESTAMP COMMENT 'Timestamp of the most recent update to this assessment; refreshed automatically on every UPDATE',
  UNIQUE KEY uq_risk_assessment_code (CODE),
  CONSTRAINT chk_likelihood CHECK (LIKELIHOOD_SCORE BETWEEN 1 AND 5),
  CONSTRAINT chk_impact      CHECK (IMPACT_SCORE     BETWEEN 1 AND 5),
  CONSTRAINT chk_velocity    CHECK (VELOCITY_SCORE IS NULL
                                    OR VELOCITY_SCORE BETWEEN 1 AND 5),
  CONSTRAINT fk_ra_risk
    FOREIGN KEY (RISK_ID)              REFERENCES ERM_RISK     (RISK_ID)     ON DELETE CASCADE,
  CONSTRAINT fk_ra_reviewer_app_user
    FOREIGN KEY (REVIEWER_APP_USER_ID) REFERENCES ERM_APP_USER (APP_USER_ID) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3 COLLATE=utf8mb3_unicode_ci
  COMMENT='Point-in-time scored evaluations of risks, capturing likelihood, impact, and residual exposure';
